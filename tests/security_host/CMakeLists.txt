cmake_minimum_required(VERSION 3.16)
project(tabremoto_vault_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_SYSROOT /Library/Developer/CommandLineTools/SDKs/MacOSX15.4.sdk)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(IDF_PATH "$ENV{HOME}/esp/esp-idf" CACHE PATH "SDK checkout")
add_subdirectory(${IDF_PATH}/components/mbedtls/mbedtls mbedtls EXCLUDE_FROM_ALL)
add_library(json STATIC ${IDF_PATH}/components/json/cJSON/cJSON.c)
target_include_directories(json PUBLIC ${IDF_PATH}/components/json/cJSON)
find_package(OpenSSL REQUIRED)
add_executable(vault_test vault_test.cpp ../../components/tabremoto_security/vault.cpp)
target_include_directories(vault_test PRIVATE ${IDF_PATH}/components/mbedtls/mbedtls/library ../../components/tabremoto_security/include)
target_link_libraries(vault_test PRIVATE mbedcrypto json OpenSSL::Crypto)
