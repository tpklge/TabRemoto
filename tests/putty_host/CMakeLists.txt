cmake_minimum_required(VERSION 3.16)
project(tabremoto_putty_adapter_tests C CXX)
set(CMAKE_OSX_SYSROOT /Library/Developer/CommandLineTools/SDKs/MacOSX15.4.sdk)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -ffunction-sections -fdata-sections -fsanitize=address,undefined")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fsanitize=address,undefined")
function(idf_component_register)
 cmake_parse_arguments(ARG "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;REQUIRES" ${ARGN})
 set(files)
 foreach(src ${ARG_SRCS})
  if(IS_ABSOLUTE "${src}")
   list(APPEND files "${src}")
  else()
   list(APPEND files "${CMAKE_CURRENT_LIST_DIR}/${src}")
  endif()
 endforeach()
 add_library(putty_adapter STATIC ${files})
 foreach(dir ${ARG_INCLUDE_DIRS} ${ARG_PRIV_INCLUDE_DIRS})
  target_include_directories(putty_adapter PUBLIC "${CMAKE_CURRENT_LIST_DIR}/${dir}")
 endforeach()
 target_include_directories(putty_adapter PUBLIC ${ARG_PRIV_INCLUDE_DIRS})
 set(COMPONENT_LIB putty_adapter PARENT_SCOPE)
endfunction()
include(${CMAKE_CURRENT_LIST_DIR}/../../components/tabremoto_putty/CMakeLists.txt)

target_link_options(ppk_test PRIVATE -Wl,-dead_strip)
target_link_libraries(ppk_test PRIVATE putty_adapter)
