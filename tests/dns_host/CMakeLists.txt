cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_SYSROOT /Library/Developer/CommandLineTools/SDKs/MacOSX15.4.sdk)
project(tabremoto_dns_host C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CARES_STATIC ON CACHE BOOL "" FORCE)
set(CARES_SHARED OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CARES_THREADS OFF CACHE BOOL "" FORCE)
set(CARES_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../research/c-ares c-ares EXCLUDE_FROM_ALL)
add_executable(dns_host dns_test.cpp ../../components/tabremoto_dns/dns_resolver.cpp)
target_include_directories(dns_host PRIVATE ../../components/tabremoto_dns/include)
target_link_libraries(dns_host PRIVATE c-ares::cares_static)
target_compile_options(dns_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(dns_host PRIVATE -fsanitize=address,undefined)
