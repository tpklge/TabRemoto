set(PUTTY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../research/putty")
set(PUTTY_CRYPTO aes-common aes-select aes-sw argon2 bcrypt blake2 blowfish ecc-arithmetic ecc-ssh hash_simple hmac mac mac_simple md5 mpint pubkey-ppk rfc6979 rsa sha256-common sha256-select sha256-sw sha512-common sha512-select sha512-sw sha3 sha1-common sha1-select sha1-sw)
set(PUTTY_UTILS base64_decode_atom base64_decode base64_encode_atom base64_encode base64_valid burnstr dupcat dupprintf dupstr marshal memory memxor ptrlen smemclr smemeq strbuf string_length_for_printf key_components)
set(PUTTY_SOURCES "${PUTTY_ROOT}/sshpubk.c" "${PUTTY_ROOT}/stubs/null-cipher.c" "${PUTTY_ROOT}/stubs/null-key.c" "${PUTTY_ROOT}/stubs/null-mac.c")
foreach(name ${PUTTY_CRYPTO})
 list(APPEND PUTTY_SOURCES "${PUTTY_ROOT}/crypto/${name}.c")
endforeach()
foreach(name ${PUTTY_UTILS})
 list(APPEND PUTTY_SOURCES "${PUTTY_ROOT}/utils/${name}.c")
endforeach()
idf_component_register(SRCS "putty_key.c" "putty_writer.c" ${PUTTY_SOURCES} INCLUDE_DIRS "include" PRIV_INCLUDE_DIRS "${PUTTY_ROOT}" "${PUTTY_ROOT}/unix" "${PUTTY_ROOT}/charset" REQUIRES esp_timer esp_system esp_psram)
target_compile_definitions(${COMPONENT_LIB} PRIVATE NO_LIBDL NOT_X_WINDOWS)
target_compile_options(${COMPONENT_LIB} PRIVATE -UNDEBUG)
